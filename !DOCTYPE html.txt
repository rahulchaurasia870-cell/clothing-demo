  <script>
    // --- Demo catalog ---
    const catalog = [
      {id:'ST-100', name:'Oversized Tee — Cloud', price:699, category:'T-Shirts', image:'https://images.unsplash.com/photo-1520975968319-53539be2a89a?q=80&w=1200&auto=format&fit=crop', desc:'Soft cotton, relaxed fit.'},
      {id:'ST-110', name:'Graphic Tee — Neon Fade', price:899, category:'T-Shirts', image:'https://images.unsplash.com/photo-1520975922284-6b456c4579a1?q=80&w=1200&auto=format&fit=crop', desc:'Pop print. Weekend-ready.'},
      {id:'DN-200', name:'Denim — Midnight Slim', price:1799, category:'Denim', image:'https://images.unsplash.com/photo-1516826957135-700deda3f9ab?q=80&w=1200&auto=format&fit=crop', desc:'Stretch denim, tapered leg.'},
      {id:'DN-210', name:'Denim — Light Wash', price:1699, category:'Denim', image:'https://images.unsplash.com/photo-1495121553079-4c61bcce1894?q=80&w=1200&auto=format&fit=crop', desc:'Casual light wash, comfy.'},
      {id:'ET-300', name:'Kurta — Indigo Weave', price:1299, category:'Ethnic', image:'https://images.unsplash.com/photo-1620641788421-7a1c342ea7d4?q=80&w=1200&auto=format&fit=crop', desc:'Breathable cotton-linen blend.'},
      {id:'ET-310', name:'Kurta — Sand Beige', price:1199, category:'Ethnic', image:'https://images.unsplash.com/photo-1591901206016-3e79fb6b3a25?q=80&w=1200&auto=format&fit=crop', desc:'Light & airy for summers.'},
      {id:'HD-400', name:'Hoodie — Charcoal Fleece', price:1499, category:'Hoodies', image:'https://images.unsplash.com/photo-1503342217505-b0a15cf70489?q=80&w=1200&auto=format&fit=crop', desc:'Cozy fleece, kangaroo pocket.'},
      {id:'AC-500', name:'Cap — Minimal Logo', price:499, category:'Accessories', image:'https://images.unsplash.com/photo-1516826957135-700deda3f9ab?q=80&w=1200&auto=format&fit=crop', desc:'Everyday essential.'}
    ];

    // --- State ---
    const state = {
      search:'',
      category:'all',
      sort:'pop',
      cart: JSON.parse(localStorage.getItem('cart')||'[]')
    };

    // --- Elements ---
    const els = {
      grid: document.getElementById('grid'),
      search: document.getElementById('search'),
      category: document.getElementById('category'),
      sort: document.getElementById('sort'),
      cartCount: document.getElementById('cartCount'),
      fabCount: document.getElementById('fabCount'),
      drawer: document.getElementById('drawer'),
      openCart: document.getElementById('openCart'),
      closeCart: document.getElementById('closeCart'),
      fab: document.getElementById('fab'),
      cartLines: document.getElementById('cartLines'),
      subtotal: document.getElementById('subtotal'),
      beginCheckoutBtn: document.getElementById('beginCheckoutBtn'),
      checkoutBox: document.getElementById('checkoutBox'),
      addShip: document.getElementById('addShip'),
      addPay: document.getElementById('addPay'),
      payNow: document.getElementById('payNow'),
      viewListAgain: document.getElementById('viewListAgain'),
      modal: document.getElementById('modal'),
      modalImg: document.getElementById('modalImg'),
      modalTitle: document.getElementById('modalTitle'),
      modalCat: document.getElementById('modalCat'),
      modalPrice: document.getElementById('modalPrice'),
      modalSku: document.getElementById('modalSku'),
      modalDesc: document.getElementById('modalDesc'),
      modalAdd: document.getElementById('modalAdd'),
      modalClose: document.getElementById('modalClose'),
    };

    // --- Utils ---
    const money = v => `₹${v.toLocaleString('en-IN')}`;
    const saveCart = () => localStorage.setItem('cart', JSON.stringify(state.cart));
    const calcSubtotal = () => state.cart.reduce((s,l)=> s + l.price*l.qty, 0);

    function renderGrid(){
      let items = catalog.slice();
      if(state.category!=='all') items = items.filter(i=> i.category===state.category);
      if(state.search) items = items.filter(i=> i.name.toLowerCase().includes(state.search.toLowerCase()));
      if(state.sort==='low') items.sort((a,b)=> a.price-b.price);
      if(state.sort==='high') items.sort((a,b)=> b.price-a.price);
      if(state.sort==='new') items.reverse();

      els.grid.innerHTML = items.map(p => `
        <article class="card" data-id="${p.id}">
          <img src="${p.image}" alt="${p.name}" />
          <div class="info">
            <div class="row" style="justify-content:space-between">
              <span class="badge">${p.category}</span>
              <span class="price">${money(p.price)}</span>
            </div>
            <strong>${p.name}</strong>
            <div class="row">
              <button class="btn add" data-id="${p.id}">Add to cart</button>
              <button class="pill view" data-id="${p.id}">View</button>
            </div>
          </div>
        </article>`).join('');

      // Fire view_item_list for currently visible items
      GA.event('view_item_list', {
        item_list_id: 'home_grid',
        item_list_name: 'Home Grid',
        items: items.map((p,idx)=> ({ item_id:p.id, item_name:p.name, item_category:p.category, price:p.price, index:idx }))
      });

      // bind buttons
      els.grid.querySelectorAll('.add').forEach(btn=> btn.addEventListener('click', ()=> addToCart(btn.dataset.id)));
      els.grid.querySelectorAll('.view').forEach(btn=> btn.addEventListener('click', ()=> openModal(btn.dataset.id)));
      els.grid.querySelectorAll('.card img, .card strong').forEach(el=> el.addEventListener('click', (e)=> {
        const id = e.currentTarget.closest('.card').dataset.id;
        selectItem(id);
        openModal(id);
      }));
    }

    function selectItem(id){
      const p = catalog.find(x=> x.id===id);
      GA.event('select_item', { item_list_id:'home_grid', items:[{ item_id:p.id, item_name:p.name, item_category:p.category, price:p.price }]});
    }

    function openModal(id){
      const p = catalog.find(x=> x.id===id);
      els.modalImg.src = p.image;
      els.modalTitle.textContent = p.name;
      els.modalCat.textContent = p.category;
      els.modalPrice.textContent = money(p.price);
      els.modalSku.textContent = p.id;
      els.modalDesc.textContent = p.desc;
      els.modal.dataset.id = id;
      els.modal.classList.add('open');
      GA.event('view_item', { items:[{ item_id:p.id, item_name:p.name, item_category:p.category, price:p.price }]});
    }

    function closeModal(){ els.modal.classList.remove('open'); }

    function addToCart(id){
      const p = catalog.find(x=> x.id===id);
      const line = state.cart.find(l=> l.id===id);
      if(line){ line.qty += 1; } else { state.cart.push({ id:p.id, name:p.name, price:p.price, image:p.image, qty:1, category:p.category }); }
      saveCart();
      renderCart();
      GA.event('add_to_cart', { currency:'INR', value:p.price, items:[{ item_id:p.id, item_name:p.name, item_category:p.category, price:p.price, quantity:1 }]});
    }

    function removeFromCart(id){
      const line = state.cart.find(l=> l.id===id);
      if(!line) return;
      line.qty -= 1;
      if(line.qty<=0) state.cart = state.cart.filter(l=> l.id!==id);
      saveCart();
      renderCart();
      GA.event('remove_from_cart', { currency:'INR', value:line.price, items:[{ item_id:line.id, item_name:line.name, item_category:line.category, price:line.price, quantity:1 }]});
    }

    function renderCart(){
      els.cartLines.innerHTML = '';
      if(state.cart.length===0){
        els.cartLines.innerHTML = `<div class="empty">Your cart is empty.</div>`;
        els.subtotal.textContent = money(0);
      } else {
        state.cart.forEach(line=> {
          const row = document.createElement('div');
          row.className = 'line';
          row.innerHTML = `
            <img src="${line.image}" alt="${line.name}" style="width:56px;height:56px;object-fit:cover;border-radius:10px;border:1px solid #273244"/>
            <div style="flex:1">
              <div style="display:flex;justify-content:space-between;gap:8px"><strong>${line.name}</strong> <span>${money(line.price)}</span></div>
              <div class="muted" style="font-size:12px">${line.id} • ${line.category}</div>
              <div class="qty" role="group" aria-label="Quantity">
                <button aria-label="decrease" onclick="removeFromCart('${line.id}')">−</button>
                <span>${line.qty}</span>
                <button aria-label="increase" onclick="addToCart('${line.id}')">+</button>
              </div>
            </div>`;
          els.cartLines.appendChild(row);
        });
        els.subtotal.textContent = money(calcSubtotal());
      }
      const count = state.cart.reduce((s,l)=> s + l.qty, 0);
      els.cartCount.textContent = count;
      els.fabCount.textContent = count;
    }

    function openCart(){ els.drawer.classList.add('open'); els.drawer.setAttribute('aria-hidden','false'); }
    function closeCart(){ els.drawer.classList.remove('open'); els.drawer.setAttribute('aria-hidden','true'); }

    // Checkout flow events
    function beginCheckout(){
      if(state.cart.length===0) return alert('Cart is empty');
      GA.event('begin_checkout', { currency:'INR', value: calcSubtotal(), items: state.cart.map(l=> ({ item_id:l.id, item_name:l.name, item_category:l.category, price:l.price, quantity:l.qty })) });
      els.checkoutBox.style.display = 'block';
    }

    function addShipping(){
      const addr = document.getElementById('shipping').value || 'Demo City';
      GA.event('add_shipping_info', { shipping_tier:'Standard', address: addr, currency:'INR', value: calcSubtotal(), items: state.cart.map(l=> ({ item_id:l.id, item_name:l.name, item_category:l.category, price:l.price, quantity:l.qty })) });
      alert('Shipping info added (demo).');
    }

    function addPayment(){
      const pay = document.getElementById('payment').value || 'UPI';
      GA.event('add_payment_info', { payment_type: pay, currency:'INR', value: calcSubtotal(), items: state.cart.map(l=> ({ item_id:l.id, item_name:l.name, item_category:l.category, price:l.price, quantity:l.qty })) });
      alert('Payment info added (demo).');
    }

    function completePurchase(){
      const txn = 'TX' + Math.floor(100000 + Math.random()*900000);
      const value = calcSubtotal();
      if(value<=0) return alert('Nothing to purchase');
      GA.event('purchase', {
        currency:'INR',
        transaction_id: txn,
        value: value,
        tax: Math.round(value*0.05),
        shipping: value>1499?0:99,
        coupon: 'NEW10',
        items: state.cart.map(l=> ({ item_id:l.id, item_name:l.name, item_category:l.category, price:l.price, quantity:l.qty }))
      });
      alert('Order placed! (demo) Transaction: '+txn);
      state.cart = [];
      saveCart();
      renderCart();
      closeCart();
    }

    // --- Wire up ---
    els.search.addEventListener('input', e=> { state.search = e.target.value; renderGrid(); });
    els.category.addEventListener('change', e=> { state.category = e.target.value; renderGrid(); });
    els.sort.addEventListener('change', e=> { state.sort = e.target.value; renderGrid(); });
    els.openCart.addEventListener('click', openCart);
    els.fab.addEventListener('click', openCart);
    els.closeCart.addEventListener('click', closeCart);
    els.beginCheckoutBtn.addEventListener('click', beginCheckout);
    els.addShip.addEventListener('click', addShipping);
    els.addPay.addEventListener('click', addPayment);
    els.payNow.addEventListener('click', completePurchase);
    els.viewListAgain.addEventListener('click', renderGrid);

    els.modal.addEventListener('click', (e)=> { if(e.target===els.modal) closeModal(); });
    els.modalClose.addEventListener('click', closeModal);
    els.modalAdd.addEventListener('click', ()=> { addToCart(els.modal.dataset.id); closeModal(); });

    // Initial render
    renderGrid();
    renderCart();
  </script>
